% Chapitre numerote : pave orange portant le numero, titre a cote,
% filet orange epais en dessous.
\titleformat{\chapter}[hang]
  {\normalfont\LARGE\bfseries\color{encre}\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000}
  {\colorbox{poly}{\parbox[c][13mm][c]{13mm}{%
     \centering\color{white}\fontsize{26}{26}\selectfont\bfseries\thechapter}}%
   \hspace{6mm}}
  {0pt}
  {}
  [{\vspace{3.5mm}{\color{poly}\titlerule[2.2pt]}}]
\titlespacing*{\chapter}{0pt}{-14pt}{22pt}

% Chapitre non numerote : titre seul, meme filet.
\titleformat{name=\chapter,numberless}[display]
  {\normalfont}
  {}
  {0pt}
  {\LARGE\bfseries\color{encre}\raggedright}
  [{\vspace{3.5mm}{\color{poly}\titlerule[2.2pt]}}]
\titlespacing*{name=\chapter,numberless}{0pt}{-14pt}{22pt}
